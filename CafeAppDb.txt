create table roles (
	id serial primary key,
	name varchar(50) not null unique
);

create table users (
	id serial primary key,
	role_id int not null references roles (id) on delete cascade,
	login varchar(50) not null,
	password varchar(50) not null,
	name varchar(50) not null,
	last_name varchar(50) not null,
	middle_name varchar(50),
	birthday date not null,
	status varchar(50) not null,
	user_photo bytea,
	contract_photo bytea
);

create table shifts (
	id serial primary key,
	shift_started timestamp not null,
	shift_ends timestamp not null
);

create table shift_employees (
	shift_id int not null references shifts (id) on delete cascade,
	user_id int not null references users (id) on delete cascade,
	primary key (shift_id, user_id)
);

create table tables (
	id serial primary key,
	number int not null
);

create table waiter_tables (
    shift_id int not null references shifts (id) on delete cascade,
	user_id int not null references users (id) on delete cascade,
	table_id int not null references tables (id) on delete cascade,
	primary key (shift_id, user_id, table_id)
);

create table orders (
	id serial primary key,
	shift_id int not null references shifts (id) on delete cascade,
	table_id int not null references tables (id) on delete cascade,
	clients_amount int not null default 1,
	content text not null,
	total_amount money not null,
	status varchar(50) not null,
	created_at time not null,
	completed_at time,
	payment_method varchar(50)
);
